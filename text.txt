def read_coder_output(latest_build)
  resources = Array(latest_build["resources"])

  metadata = resources.find { |r|
    r["name"] == "coder_output"
  }&.dig("metadata")

  if Array(metadata).empty?
    workspace = resources.find { |r|
      r["name"] == "workspace" &&
      r["type"] == "docker_container"
    }

    candidate = Array(workspace&.dig("metadata"))
    keys = candidate.map { |m| m["key"] }

    if %w[floating_ip port].all? { |k| keys.include?(k) }
      metadata = candidate
    end
  end

  Array(metadata).to_h { |m|
    [m["key"].to_sym, m["value"]]
  }
end
