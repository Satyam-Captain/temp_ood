BASE="[SHARED_APP_ROOT]/ood-gnome"

cat > "$BASE/bin/desktopctl.v2" <<'EOF'
#!/usr/bin/env bash
set -euo pipefail
umask 077

STATE_DIR="$HOME/.local/state/ood-gnome-v2"
STATE_FILE="$STATE_DIR/allocation"
SS=/usr/sbin/ss

mkdir -p "$STATE_DIR"
chmod 700 "$STATE_DIR"

show_status() {
  if [[ ! -f "$STATE_FILE" ]]; then
    echo "NO_ALLOCATION"
    return
  fi

  read -r display port < "$STATE_FILE"

  echo "USER=$(id -un)"
  echo "UID=$(id -u)"
  echo "DISPLAY=:$display"
  echo "VNC_PORT=$port"
  echo "STATE=ALLOCATED_ONLY"
}

case "${1:-}" in

  allocate)
    if ! mkdir -m 700 "$STATE_DIR/.allocating" 2>/dev/null; then
      echo "Another allocation is in progress or a stale lock exists"
      exit 1
    fi

    trap 'rmdir "$STATE_DIR/.allocating" 2>/dev/null || true' EXIT

    if [[ -f "$STATE_FILE" ]]; then
      show_status
      exit 0
    fi

    for display in $(seq 40 79); do
      port=$((5900 + display))
      lock="/tmp/ood-gnome-v2-display-$display"

      # Atomic reservation across participating users
      if ! mkdir -m 700 "$lock" 2>/dev/null; then
        continue
      fi

      # Check for an existing X11 display or TCP listener
      if [[ -e "/tmp/.X${display}-lock" ||
            -e "/tmp/.X11-unix/X${display}" ]] ||
         "$SS" -H -ltn "( sport = :$port )" | grep -q .; then
        rmdir "$lock"
        continue
      fi

      printf '%s %s\n' "$display" "$port" > "$STATE_FILE"

      echo "ALLOCATION_SUCCESS"
      show_status
      exit 0
    done

    echo "No display available in the test range"
    exit 1
    ;;

  status)
    show_status
    ;;

  release)
    if [[ ! -f "$STATE_FILE" ]]; then
      echo "NO_ALLOCATION"
      exit 0
    fi

    read -r display port < "$STATE_FILE"

    [[ "$display" =~ ^[0-9]+$ ]] || exit 1

    lock="/tmp/ood-gnome-v2-display-$display"

    # Never release a display while it is in use
    if [[ -e "/tmp/.X${display}-lock" ||
          -e "/tmp/.X11-unix/X${display}" ]] ||
       "$SS" -H -ltn "( sport = :$port )" | grep -q .; then
      echo "Display still in use; stop its server first"
      exit 1
    fi

    if [[ -d "$lock" ]] &&
       [[ "$(stat -c %u "$lock")" == "$(id -u)" ]]; then
      rmdir "$lock"
      rm -f "$STATE_FILE"
      echo "ALLOCATION_RELEASED"
    else
      echo "Reservation ownership mismatch"
      exit 1
    fi
    ;;

  *)
    echo "Usage: desktopctl.v2 {allocate|status|release}"
    exit 1
    ;;
esac
EOF

chmod 755 "$BASE/bin/desktopctl.v2"

bash -n "$BASE/bin/desktopctl.v2" \
  && echo "Syntax OK"
