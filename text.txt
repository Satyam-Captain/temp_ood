cat > config.ru <<'EOF'
require 'open3'
require 'cgi'
require 'securerandom'
require 'yaml'

REMOTE = '[DESKTOP_USER]@[DESKTOP_HOST]'
TOKEN = SecureRandom.hex(32)

def desktop_command(action)
  return [false, "Invalid action"] unless
    %w[start status stop].include?(action)

  out, err, status = Open3.capture3(
    'ssh',
    '-T',
    '-o', 'BatchMode=yes',
    '-o', 'StrictHostKeyChecking=yes',
    '-o', 'ConnectTimeout=5',
    REMOTE,
    "\"$HOME/ood-desktop-poc/bin/desktopctl\" #{action}"
  )

  [status.success?, status.success? ? out : err]
rescue StandardError => e
  [false, "#{e.class}: #{e.message}"]
end

run lambda { |env|
  method = env['REQUEST_METHOD']
  message = "Desktop status"

  if method == 'POST'
    params = CGI.parse(env['rack.input'].read(1024))
    action = params.fetch('action', []).first
    csrf = params.fetch('csrf', []).first

    unless csrf == TOKEN && %w[start stop].include?(action)
      next [
        403,
        {'Content-Type' => 'text/plain'},
        ['Invalid request']
      ]
    end

    ok, result = desktop_command(action)
    message = "#{action.upcase}: #{ok ? 'SUCCESS' : 'FAILED'}"
  end

  ok, status_text = desktop_command('status')

  connect_url = begin
    manifest = File.expand_path(
      '../remote_gnome_poc/manifest.yml', __dir__
    )
    YAML.safe_load(File.read(manifest)).fetch('url')
  rescue StandardError
    ''
  end

  connect_link = if connect_url.empty?
    'Connection shortcut unavailable'
  else
    "<a href=\"#{CGI.escapeHTML(connect_url)}\" " \
    "target=\"_blank\" rel=\"noopener noreferrer\">" \
    "Connect to Desktop</a>"
  end

  html = <<~HTML
    <h2>Remote GNOME Desktop Launcher</h2>
    <p>#{CGI.escapeHTML(message)}</p>

    <form method="post">
      <input type="hidden" name="csrf" value="#{TOKEN}">
      <button type="submit" name="action" value="start">
        Launch Desktop
      </button>
      <button type="submit" name="action" value="stop">
        Stop Desktop
      </button>
    </form>

    <p>#{connect_link}</p>

    <h3>Session Status</h3>
    <pre>#{CGI.escapeHTML(status_text)}</pre>

    <p>Refresh this page to update the status.</p>
  HTML

  [
    200,
    {
      'Content-Type' => 'text/html; charset=utf-8',
      'Cache-Control' => 'no-store'
    },
    [html]
  ]
}
EOF
