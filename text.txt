  printf '%s\n' "$$" > "$ROOT/worker.pid"

  "$XVNC" ":$DISPLAY_NO" -geometry 1280x800 -depth 24 \
    -localhost -SecurityTypes VncAuth -rfbauth "$PASSFILE" \
    -rfbport "$VNC_PORT" > "$LOGS/xvnc.log" 2>&1 &
  XVNC_PID=$!
  printf '%s\n' "$XVNC_PID" > "$ROOT/xvnc.pid"
  sleep 2
  kill -0 "$XVNC_PID" 2>/dev/null || { echo 'Xvnc exited; inspect xvnc.log'; return 1; }

  DISPLAY=":$DISPLAY_NO" XDG_SESSION_TYPE=x11 \
    "$DBUS_RUN" -- "$GNOME" > "$LOGS/gnome.log" 2>&1 &
  GNOME_PID=$!
  printf '%s\n' "$GNOME_PID" > "$ROOT/gnome.pid"

  "$WEBSOCKIFY_RUN" "$WS_BIND_IP:$WS_PORT" "127.0.0.1:$VNC_PORT" \
    > "$LOGS/websockify.log" 2>&1 &
  WS_PID=$!
  printf '%s\n' "$WS_PID" > "$ROOT/websockify.pid"
  sleep 2
  kill -0 "$WS_PID" 2>/dev/null || { echo 'websockify exited; inspect websockify.log'; return 1; }

  while kill -0 "$XVNC_PID" 2>/dev/null && \
        kill -0 "$GNOME_PID" 2>/dev/null && \
        kill -0 "$WS_PID" 2>/dev/null; do
    sleep 3
  done
  echo 'One desktop component exited; stopping the others'
}

# The stop command only signals PIDs belonging to the current user and matching
# our own known command lines. It never uses a broad pkill pattern.
stop_owned_pid() {
  local file=$1 match=$2 pid args uid
  [[ -f "$file" ]] || return 0
  read -r pid < "$file" || return 0
  [[ "$pid" =~ ^[0-9]+$ ]] || return 0
  uid=$(ps -p "$pid" -o uid= 2>/dev/null | tr -d ' ' || true)
  args=$(ps -p "$pid" -o args= 2>/dev/null || true)
  if [[ "$uid" == "$(id -u)" && "$args" == *"$match"* ]]; then
    kill -TERM "$pid" 2>/dev/null || true
  fi
}

release_reservations() {
  if load_allocation; then
    if listening "$VNC_PORT" || [[ -e "/tmp/.X11-unix/X$DISPLAY_NO" || -e "/tmp/.X${DISPLAY_NO}-lock" ]]; then
      echo 'VNC display still in use; allocation retained'; return 1
    fi
  fi
  if load_ws && listening "$WS_PORT"; then
    echo 'WebSocket port still in use; allocation retained'; return 1
  fi
  if load_allocation; then
    owned_rmdir "$(display_lock "$DISPLAY_NO")"
    rm -f "$ALLOC"
  fi
  if load_ws; then
    owned_rmdir "$(ws_lock "$WS_PORT")"
    rm -f "$WS_ALLOC"
  fi
  rm -f "$ROOT/xvnc.pid" "$ROOT/gnome.pid" "$ROOT/websockify.pid" "$ROOT/worker.pid"
  echo 'RESERVATIONS_RELEASED'
}

case "${1:-}" in
  start)
    check_tools
    if "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
      echo 'DESKTOP_ALREADY_RUNNING'
      show_status
      exit 0
    fi
    mkdir -m 700 "$ROOT/.starting" 2>/dev/null || { echo 'Concurrent start or stale start lock'; exit 1; }
    trap 'rmdir "$ROOT/.starting" 2>/dev/null || true' EXIT
    reserve_display
    if listening "$VNC_PORT" || [[ -e "/tmp/.X${DISPLAY_NO}-lock" || -e "/tmp/.X11-unix/X${DISPLAY_NO}" ]]; then
      echo 'Reserved VNC display is already in use; refusing to alter its credentials'
      exit 1
    fi
    reserve_ws
    python3 -c 'import secrets,string; a=string.ascii_letters+string.digits; print("".join(secrets.choice(a) for _ in range(8)))' > "$PASS"
    "$VNCPASSWD" -f < "$PASS" > "$PASSFILE"
    chmod 600 "$PASS" "$PASSFILE"
    printf 'host: %s\nport: %s\ndisplay: %s\nwebsocket: %s\n' \
      "$(hostname -f)" "$VNC_PORT" "$DISPLAY_NO" "$WS_PORT" > "$ROOT/connection.yml"
    chmod 600 "$ROOT/connection.yml"
    # tmux 1.8 accepts the shell command as one argument.
    "$TMUX" -L "$SOCKET" new-session -d -s "$SESSION" \
      "/bin/bash \"$SELF\" worker >> \"$LOGS/worker.log\" 2>&1"
    for _ in $(seq 1 25); do
      if listening "$VNC_PORT" && listening "$WS_PORT"; then
        echo 'DESKTOP_NETWORK_READY (GNOME may still be initializing)'
        show_status
        exit 0
      fi
      if ! "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
        echo "Desktop worker exited. Inspect $LOGS/worker.log"; exit 1
      fi
      sleep 1
    done
    echo "Desktop has not reached network-ready state. Inspect $LOGS/worker.log"
    exit 1
    ;;
  status) show_status ;;
  stop)
    # Do not affect another user's tmux server or desktop processes.
    stop_owned_pid "$ROOT/worker.pid" "$SELF"
    sleep 2
    "$TMUX" -L "$SOCKET" kill-session -t "$SESSION" 2>/dev/null || true
    load_allocation || true
    load_ws || true
    [[ ! -f "$ROOT/websockify.pid" ]] || stop_owned_pid "$ROOT/websockify.pid" "${WS_PORT:-__unmatched__}"
    [[ ! -f "$ROOT/gnome.pid" ]] || stop_owned_pid "$ROOT/gnome.pid" 'gnome-session'
    [[ ! -f "$ROOT/xvnc.pid" ]] || stop_owned_pid "$ROOT/xvnc.pid" ":${DISPLAY_NO:-__unmatched__}"
    sleep 2
    release_reservations
    ;;
  release)
    if "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
      echo 'Desktop session is active; run stop first'; exit 1
    fi
    release_reservations
    ;;
  *) echo 'Usage: desktopctl.v2 {start|status|stop|release}'; exit 2 ;;
esac
