python3 - <<'PY'
from pathlib import Path
import shutil
import ipaddress

p = Path.home() / "ood-desktop-poc/bin/desktopctl"
s = p.read_text()

if "WEB_RUN=" in s:
    raise SystemExit("Already patched. No changes made.")

web_run = input("Absolute path to websockify run: ").strip()
web_bind = input("Approved desktop IPv4 address: ").strip()

if not Path(web_run).is_file() or '"' in web_run:
    raise SystemExit("Invalid websockify launcher path.")

ipaddress.IPv4Address(web_bind)

def replace_once(old, new):
    global s
    if s.count(old) != 1:
        raise SystemExit("Unexpected desktopctl version. No changes saved.")
    s = s.replace(old, new, 1)

replace_once(
    "VNC_PORT=5930\n",
    f'''VNC_PORT=5930
WEB_RUN="{web_run}"
WEB_BIND="{web_bind}"
WEB_PORT=6081
'''
)

replace_once(
    '  if [[ -n "${GNOME_PID:-}" ]]; then',
    '''  if [[ -n "${WEB_PID:-}" ]]; then
    kill -TERM "$WEB_PID" 2>/dev/null || true
  fi
  if [[ -n "${GNOME_PID:-}" ]]; then'''
)

replace_once(
    "    python3 -c '\nimport secrets, string",
    '''    [[ -x "$WEB_RUN" ]] || {
      echo "websockify launcher not executable"
      exit 1
    }

    if ss -ltn "( sport = :$WEB_PORT )" |
       grep -q '^LISTEN'; then
      echo "WebSocket port already occupied"
      exit 1
    fi

    python3 -c '
import secrets, string'''
)

replace_once(
    '''    echo "$GNOME_PID" > "$STATE/gnome.pid"

    wait "$GNOME_PID"''',
    '''    echo "$GNOME_PID" > "$STATE/gnome.pid"

    "$WEB_RUN" "$WEB_BIND:$WEB_PORT" "127.0.0.1:$VNC_PORT" \\
      > "$LOGS/websockify30.log" 2>&1 &

    WEB_PID=$!
    echo "$WEB_PID" > "$STATE/websockify.pid"

    sleep 2
    kill -0 "$WEB_PID" 2>/dev/null || {
      echo "websockify startup failed" >&2
      exit 1
    }

    wait "$GNOME_PID"'''
)

replace_once(
    "    ss -ltn '( sport = :5930 )'",
    '''    echo "=== Xvnc ==="
    ss -ltn '( sport = :5930 )'
    echo "=== websockify ==="
    ss -ltn '( sport = :6081 )' '''
)

replace_once(
    '''    for entry in "gnome:gnome.pid" "Xvnc:xvnc.pid"; do''',
    '''    if [[ -f "$STATE/websockify.pid" ]]; then
      pid=$(cat "$STATE/websockify.pid")
      if [[ "$pid" =~ ^[0-9]+$ ]]; then
        args=$(ps -p "$pid" -o args= 2>/dev/null || true)
        if [[ "$args" == *":6081"* &&
              "$args" == *":5930"* ]]; then
          kill -TERM "$pid" 2>/dev/null || true
        fi
      fi
    fi

    for entry in "gnome:gnome.pid" "Xvnc:xvnc.pid"; do'''
)

backup = p.with_name("desktopctl.before-websockify.bak")
if backup.exists():
    raise SystemExit("Backup already exists; no changes saved.")

shutil.copy2(p, backup)
p.write_text(s)
print("Desktop launcher updated.")
PY

from pathlib import Path
import shutil

p = Path("config.ru")
s = p.read_text()

needle = "  connect_link = if connect_url.empty?\n"

if s.count(needle) != 1 or s.count("require 'yaml'\n") != 1:
    raise SystemExit("Unexpected config.ru structure. No changes made.")

s = s.replace(
    "require 'yaml'\n",
    "require 'yaml'\nrequire 'uri'\n",
    1
)

addition = r'''  if ok && status_text.include?('TMUX_SESSION_RUNNING') &&
     !connect_url.empty?

    output, _, result = Open3.capture3(
      'ssh', '-T',
      '-o', 'BatchMode=yes',
      '-o', 'StrictHostKeyChecking=yes',
      '-o', 'ConnectTimeout=5',
      REMOTE,
      'cat "$HOME/ood-desktop-poc/sessions/desktop30/password"'
    )

    password = output.lines.map(&:strip).find do |line|
      line.match?(/\A[A-Za-z0-9]{8}\z/)
    end

    if result.success? && password
      connect_url = "#{connect_url.split('#').first}#password=#{URI.encode_www_form_component(password)}"
    else
      connect_url = ''
    end
  else
    connect_url = ''
  end

'''

s = s.replace(needle, addition + needle, 1)

backup = Path("config.ru.before-auto-password.bak")
if backup.exists():
    raise SystemExit("Backup already exists. No changes made.")

shutil.copy2(p, backup)
p.write_text(s)

print("Automatic VNC password integration added.")
PY
