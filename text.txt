#!/usr/bin/env bash
# OOD GNOME multi-user POC: one independent desktop per Linux account.
# Must be executed as the target user, never via sudo/root.
set -Eeuo pipefail
umask 077

# CONFIGURE ONCE. Use only an approved network-facing WebSocket listener.
WEBSOCKIFY_RUN='[WEBSOCKIFY_RUN_ABSOLUTE_PATH]'
WS_BIND_IP='[DESKTOP_PRIVATE_IP]'
WS_PORTS=(6081)  # Add only further ports explicitly approved for OOD -> desktop.

XVNC=/usr/bin/Xvnc
VNCPASSWD=/usr/bin/vncpasswd
SS=/usr/sbin/ss
TMUX=/usr/bin/tmux
DBUS_RUN=/usr/bin/dbus-run-session
GNOME=/usr/bin/gnome-session
SELF=$(readlink -f -- "$0")

ROOT="$HOME/.local/state/ood-gnome-v2"
LOGS="$ROOT/logs"
ALLOC="$ROOT/allocation"              # Existing v2 format: display vnc-port
WS_ALLOC="$ROOT/websocket_port"
PASS="$ROOT/password"                 # Plain VNC password, never share
PASSFILE="$ROOT/vnc.passwd"           # VNC authentication file
SOCKET=ood_gnome_v2
SESSION=desktop
XVNC_PID=''
GNOME_PID=''
WS_PID=''

mkdir -p -m 700 "$ROOT" "$LOGS"
chmod 700 "$ROOT" "$LOGS"
[[ $(id -u) -ne 0 ]] || { echo 'Refusing to run as root'; exit 1; }

listening() {
  "$SS" -H -ltn "( sport = :$1 )" | grep -q .
}

owned_rmdir() {
  local dir=$1
  if [[ -d "$dir" ]] && [[ $(stat -c '%u' -- "$dir") == "$(id -u)" ]]; then
    rmdir -- "$dir" 2>/dev/null || true
  fi
}

display_lock() { printf '/tmp/ood-gnome-v2-display-%s' "$1"; }
ws_lock() { printf '/tmp/ood-gnome-v2-websocket-%s' "$1"; }

load_allocation() {
  [[ -f "$ALLOC" ]] || return 1
  read -r DISPLAY_NO VNC_PORT < "$ALLOC" || return 1
  [[ "$DISPLAY_NO" =~ ^[0-9]+$ && "$VNC_PORT" =~ ^[0-9]+$ ]] || return 1
  (( DISPLAY_NO >= 40 && DISPLAY_NO <= 79 )) || return 1
  (( VNC_PORT == DISPLAY_NO + 5900 )) || return 1
}

load_ws() {
  [[ -f "$WS_ALLOC" ]] || return 1
  read -r WS_PORT < "$WS_ALLOC" || return 1
  [[ "$WS_PORT" =~ ^[0-9]+$ ]] || return 1
}

check_tools() {
  local cmd
  for cmd in "$XVNC" "$VNCPASSWD" "$SS" "$TMUX" "$DBUS_RUN" "$GNOME" "$WEBSOCKIFY_RUN"; do
    [[ -x "$cmd" ]] || { echo "Missing/not executable: $cmd"; return 1; }
  done
  [[ "$WS_BIND_IP" != '[DESKTOP_PRIVATE_IP]' ]] || {
    echo 'Configure the approved WS_BIND_IP first'; return 1;
  }
  [[ "$WEBSOCKIFY_RUN" != '[WEBSOCKIFY_RUN_ABSOLUTE_PATH]' ]] || {
    echo 'Configure WEBSOCKIFY_RUN first'; return 1;
  }
}

reserve_display() {
  if [[ -f "$ALLOC" ]]; then
    load_allocation || { echo 'Invalid existing allocation; inspect privately'; return 1; }
    [[ $(stat -c '%u' -- "$(display_lock "$DISPLAY_NO")" 2>/dev/null || echo missing) == "$(id -u)" ]] || {
      echo 'Existing display reservation missing or not owned by this user'; return 1;
    }
    return 0
  fi
  local d p lock
  for d in $(seq 40 79); do
    p=$((5900+d))
    lock=$(display_lock "$d")
    mkdir -m 700 -- "$lock" 2>/dev/null || continue
    if [[ -e "/tmp/.X${d}-lock" || -e "/tmp/.X11-unix/X${d}" ]] || listening "$p"; then
      owned_rmdir "$lock"
      continue
    fi
    DISPLAY_NO=$d
    VNC_PORT=$p
    printf '%s %s\n' "$d" "$p" > "$ALLOC"
    return 0
  done
  echo 'No free X display in the configured range'; return 1
}

reserve_ws() {
  if [[ -f "$WS_ALLOC" ]]; then
    load_ws || { echo 'Invalid existing WebSocket reservation'; return 1; }
    [[ $(stat -c '%u' -- "$(ws_lock "$WS_PORT")" 2>/dev/null || echo missing) == "$(id -u)" ]] || {
      echo 'Existing WebSocket reservation missing or not owned'; return 1;
    }
    return 0
  fi
  local p lock
  for p in "${WS_PORTS[@]}"; do
    [[ "$p" =~ ^[0-9]+$ ]] || continue
    lock=$(ws_lock "$p")
    mkdir -m 700 -- "$lock" 2>/dev/null || continue
    if listening "$p"; then
      owned_rmdir "$lock"
      continue
    fi
    WS_PORT=$p
    printf '%s\n' "$p" > "$WS_ALLOC"
    return 0
  done
  echo 'No approved WebSocket port available. Do not choose unapproved ports.'
  return 1
}

show_status() {
  if "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
    echo 'TMUX_SESSION_RUNNING'
  else
    echo 'TMUX_SESSION_NOT_RUNNING'
  fi
  if load_allocation; then
    echo "DISPLAY=:$DISPLAY_NO VNC_PORT=$VNC_PORT"
    if listening "$VNC_PORT"; then echo 'XVNC_LISTENING'; else echo 'XVNC_NOT_LISTENING'; fi
  else
    echo 'NO_DISPLAY_ALLOCATION'
  fi
  if load_ws; then
    echo "WEBSOCKET_PORT=$WS_PORT"
    if listening "$WS_PORT"; then echo 'WEBSOCKET_LISTENING'; else echo 'WEBSOCKET_NOT_LISTENING'; fi
  else
    echo 'NO_WEBSOCKET_ALLOCATION'
  fi
}

# Signals and normal exit clean up only the subprocesses owned by this worker.
worker_cleanup() {
  trap - EXIT HUP INT TERM
  [[ -z "$WS_PID" ]] || kill -TERM "$WS_PID" 2>/dev/null || true
  [[ -z "$GNOME_PID" ]] || kill -TERM "$GNOME_PID" 2>/dev/null || true
  [[ -z "$XVNC_PID" ]] || kill -TERM "$XVNC_PID" 2>/dev/null || true
  rm -f "$ROOT/worker.pid"
}

# Called only by this user's detached tmux session.
worker() {
  check_tools
  load_allocation && load_ws || { echo 'Missing resource allocation'; exit 1; }
  trap worker_cleanup EXIT HUP INT TERM
  printf '%s\n' "$$" > "$ROOT/worker.pid"

  "$XVNC" ":$DISPLAY_NO" -geometry 1280x800 -depth 24 \
    -localhost -SecurityTypes VncAuth -rfbauth "$PASSFILE" \
    -rfbport "$VNC_PORT" > "$LOGS/xvnc.log" 2>&1 &
  XVNC_PID=$!
  printf '%s\n' "$XVNC_PID" > "$ROOT/xvnc.pid"
  sleep 2
  kill -0 "$XVNC_PID" 2>/dev/null || { echo 'Xvnc exited; inspect xvnc.log'; return 1; }

  DISPLAY=":$DISPLAY_NO" XDG_SESSION_TYPE=x11 \
    "$DBUS_RUN" -- "$GNOME" > "$LOGS/gnome.log" 2>&1 &
  GNOME_PID=$!
  printf '%s\n' "$GNOME_PID" > "$ROOT/gnome.pid"

  "$WEBSOCKIFY_RUN" "$WS_BIND_IP:$WS_PORT" "127.0.0.1:$VNC_PORT" \
    > "$LOGS/websockify.log" 2>&1 &
  WS_PID=$!
  printf '%s\n' "$WS_PID" > "$ROOT/websockify.pid"
  sleep 2
  kill -0 "$WS_PID" 2>/dev/null || { echo 'websockify exited; inspect websockify.log'; return 1; }

  while kill -0 "$XVNC_PID" 2>/dev/null && \
        kill -0 "$GNOME_PID" 2>/dev/null && \
        kill -0 "$WS_PID" 2>/dev/null; do
    sleep 3
  done
  echo 'One desktop component exited; stopping the others'
}

# The stop command only signals PIDs belonging to the current user and matching
# our own known command lines. It never uses a broad pkill pattern.
stop_owned_pid() {
  local file=$1 match=$2 pid args uid
  [[ -f "$file" ]] || return 0
  read -r pid < "$file" || return 0
  [[ "$pid" =~ ^[0-9]+$ ]] || return 0
  uid=$(ps -p "$pid" -o uid= 2>/dev/null | tr -d ' ' || true)
  args=$(ps -p "$pid" -o args= 2>/dev/null || true)
  if [[ "$uid" == "$(id -u)" && "$args" == *"$match"* ]]; then
    kill -TERM "$pid" 2>/dev/null || true
  fi
}

release_reservations() {
  if load_allocation; then
    if listening "$VNC_PORT" || [[ -e "/tmp/.X11-unix/X$DISPLAY_NO" || -e "/tmp/.X${DISPLAY_NO}-lock" ]]; then
      echo 'VNC display still in use; allocation retained'; return 1
    fi
  fi
  if load_ws && listening "$WS_PORT"; then
    echo 'WebSocket port still in use; allocation retained'; return 1
  fi
  if load_allocation; then
    owned_rmdir "$(display_lock "$DISPLAY_NO")"
    rm -f "$ALLOC"
  fi
  if load_ws; then
    owned_rmdir "$(ws_lock "$WS_PORT")"
    rm -f "$WS_ALLOC"
  fi
  rm -f "$ROOT/xvnc.pid" "$ROOT/gnome.pid" "$ROOT/websockify.pid" "$ROOT/worker.pid"
  echo 'RESERVATIONS_RELEASED'
}

case "${1:-}" in
  start)
    check_tools
    if "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
      echo 'DESKTOP_ALREADY_RUNNING'
      show_status
      exit 0
    fi
    mkdir -m 700 "$ROOT/.starting" 2>/dev/null || { echo 'Concurrent start or stale start lock'; exit 1; }
    trap 'rmdir "$ROOT/.starting" 2>/dev/null || true' EXIT
    reserve_display
    if listening "$VNC_PORT" || [[ -e "/tmp/.X${DISPLAY_NO}-lock" || -e "/tmp/.X11-unix/X${DISPLAY_NO}" ]]; then
      echo 'Reserved VNC display is already in use; refusing to alter its credentials'
      exit 1
    fi
    reserve_ws
    python3 -c 'import secrets,string; a=string.ascii_letters+string.digits; print("".join(secrets.choice(a) for _ in range(8)))' > "$PASS"
    "$VNCPASSWD" -f < "$PASS" > "$PASSFILE"
    chmod 600 "$PASS" "$PASSFILE"
    printf 'host: %s\nport: %s\ndisplay: %s\nwebsocket: %s\n' \
      "$(hostname -f)" "$VNC_PORT" "$DISPLAY_NO" "$WS_PORT" > "$ROOT/connection.yml"
    chmod 600 "$ROOT/connection.yml"
    # tmux 1.8 accepts the shell command as one argument.
    "$TMUX" -L "$SOCKET" new-session -d -s "$SESSION" \
      "/bin/bash \"$SELF\" worker >> \"$LOGS/worker.log\" 2>&1"
    for _ in $(seq 1 25); do
      if listening "$VNC_PORT" && listening "$WS_PORT"; then
        echo 'DESKTOP_NETWORK_READY (GNOME may still be initializing)'
        show_status
        exit 0
      fi
      if ! "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
        echo "Desktop worker exited. Inspect $LOGS/worker.log"; exit 1
      fi
      sleep 1
    done
    echo "Desktop has not reached network-ready state. Inspect $LOGS/worker.log"
    exit 1
    ;;
  status) show_status ;;
  stop)
    # Do not affect another user's tmux server or desktop processes.
    stop_owned_pid "$ROOT/worker.pid" "$SELF"
    sleep 2
    "$TMUX" -L "$SOCKET" kill-session -t "$SESSION" 2>/dev/null || true
    load_allocation || true
    load_ws || true
    [[ ! -f "$ROOT/websockify.pid" ]] || stop_owned_pid "$ROOT/websockify.pid" "${WS_PORT:-__unmatched__}"
    [[ ! -f "$ROOT/gnome.pid" ]] || stop_owned_pid "$ROOT/gnome.pid" 'gnome-session'
    [[ ! -f "$ROOT/xvnc.pid" ]] || stop_owned_pid "$ROOT/xvnc.pid" ":${DISPLAY_NO:-__unmatched__}"
    sleep 2
    release_reservations
    ;;
  release)
    if "$TMUX" -L "$SOCKET" has-session -t "$SESSION" 2>/dev/null; then
      echo 'Desktop session is active; run stop first'; exit 1
    fi
    release_reservations
    ;;
  *) echo 'Usage: desktopctl.v2 {start|status|stop|release}'; exit 2 ;;
esac
