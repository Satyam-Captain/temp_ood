APP="/var/www/ood/apps/sys/remote_gnome_multi"

sudo cp -p "$APP/config.ru" "$APP/config.ru.before-regex-fix"

sudo python3 - <<'PY'
from pathlib import Path

p = Path("/var/www/ood/apps/sys/remote_gnome_multi/config.ru")
s = p.read_text()

wrong = r"/\\A[0-9a-f]{64}\\z/"
correct = r"/\A[0-9a-f]{64}\z/"

if s.count(wrong) != 1:
    raise SystemExit("Expected escaping error not found; file unchanged.")

p.write_text(s.replace(wrong, correct, 1))
print("CSRF regex corrected")
PY

ruby -c "$APP/config.ru"
